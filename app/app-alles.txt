from app.controllers.turntable_controller import TurntableController

turntable_controller = None

def datetime_filter(value, format='%d.%m.%Y %H:%M'):
    """
    Formatiert einen Timestamp oder datetime-Objekt
    
    Args:
        value: Timestamp oder datetime-Objekt
        format: Ausgabeformat für das Datum
    
    Returns:
        Formatierter Datums-String
    """
    if value is None:
        return ''
    
    # Wenn es sich um einen Timestamp handelt
    if isinstance(value, (int, float)):
        value = datetime.datetime.fromtimestamp(value)
    
    # Wenn es sich um ein datetime-Objekt handelt
    if isinstance(value, datetime.datetime):
        return value.strftime(format)
    
    return str(value)

def create_app(config_name=None):
    """
    Erstellt und konfiguriert die Flask-Anwendung

    Args:
        config_name: Name der Konfiguration ('development', 'production', 'testing')

    Returns:
        Flask-App-Instanz
    """
    app = Flask(__name__,
                static_folder="../static",
                static_url_path='/static',
                template_folder="../templates")

    # Konfiguration laden
    app.config.from_object('app.config.DevelopmentConfig')
    if config_name == 'production':
        app.config.from_object('app.config.ProductionConfig')
    elif config_name == 'testing':
        app.config.from_object('app.config.TestingConfig')

    # Jinja2-Filter registrieren
    app.jinja_env.filters['datetime'] = datetime_filter

    # Stelle sicher, dass Verzeichnisse existieren
    create_directories()

    # Controller initialisieren
    initialize_controllers()

    # Logging konfigurieren
    configure_logging(app)

    # Blueprints registrieren
    register_blueprints(app)

    # Sicherheits-Header hinzufügen
    @app.after_request
    def add_security_headers(response):
        """Fügt Sicherheits-Header zu allen Antworten hinzu"""
        # Schutz vor Clickjacking
        response.headers['X-Frame-Options'] = 'SAMEORIGIN'
        # Aktivieren der XSS-Filterung im Browser
        response.headers['X-XSS-Protection'] = '1; mode=block'
        # Verhindern von MIME-Typ-Sniffing
        response.headers['X-Content-Type-Options'] = 'nosniff'
        # Content-Security-Policy - weniger restriktiv für Entwicklung
        response.headers['Content-Security-Policy'] = (
            "default-src 'self'; "
            "script-src 'self' 'unsafe-inline' 'unsafe-eval'; "
            "style-src 'self' 'unsafe-inline'; "
            "img-src 'self' data:; "
            "connect-src 'self'"
        )
        return response

    return app

def create_directories():

    for root, dirs, files in os.walk(verzeichnis):
        # Entferne .git aus der Liste der zu durchsuchenden Verzeichnisse
        if '.git' in dirs:
            dirs.remove('.git')
        if 'bin' in dirs:
            dirs.remove('bin')
        if '__pycache__' in dirs:
            dirs.remove('__pycache__')
        if '.idea' in dirs: 
            dirs.remove('.idea')
        if 'myenv' in dirs:
            dirs.remove('myenv')
        for file in files:
            dateipfad = os.path.join(root, file)
            print(f"\n--- Datei: {dateipfad} ---")
            try:
                # Versuche, die Datei im UTF-8 Format zu öffnen
                with open(dateipfad, 'r', encoding='utf-8') as f:
                    inhalt = f.read()
                    print(inhalt)

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
app/utils/config_manager.py
Verwaltet die Benutzerkonfiguration
"""

import os
import json
import logging

class ConfigManager:
    """Klasse zur Verwaltung der Benutzerkonfiguration"""
    
    DEFAULT_CONFIG = {
        'camera': {
            'device_path': '/dev/video0',
            'type': 'webcam',
            'resolution': {
                'width': 1280,
                'height': 720
            }
        },
        'arduino': {
            'port': '/dev/ttyACM0',
            'baudrate': 9600
        },
        'rotation': {
            'default_degrees': 15,
            'default_interval': 5
        },
        'simulator': {
            'enabled': True
        }
    }

    def __init__(self, config_path=None):
        """
        Initialisiert die Konfigurationsverwaltung
        
        Args:
            config_path: Pfad zur Konfigurationsdatei (optional)
        """
        # Logger einrichten
        self.logger = logging.getLogger(__name__)
        
        # Projektverzeichnis bestimmen
        self.project_dir = os.path.abspath(os.path.dirname(os.path.dirname(os.path.dirname(__file__))))

        # Konfigurationspfad festlegen
        if config_path is None:
            config_path = os.path.join(self.project_dir, 'config.json')
        self.config_path = config_path
        
        # Konfiguration laden
        self.config = self.load_config()
        self.logger.debug(f"ConfigManager initialisiert mit Konfigurationspfad: {self.config_path}")

    def load_config(self):
        """
        Lädt die Konfiguration aus der Datei oder erstellt die Standardkonfiguration
        
        Returns:
            Konfiguration als Dictionary
        """
        try:
            # Stelle sicher, dass das Verzeichnis existiert
            os.makedirs(os.path.dirname(self.config_path), exist_ok=True)

            # Versuche, bestehende Konfiguration zu laden
            if os.path.exists(self.config_path):
                with open(self.config_path, 'r') as f:
                    config = json.load(f)
                    # Mit Standardkonfiguration zusammenführen, um sicherzustellen,
                    # dass alle Schlüssel existieren
                    merged_config = self._merge_configs(self.DEFAULT_CONFIG, config)
                    self.logger.debug("Konfiguration aus Datei geladen und zusammengeführt")
                    return merged_config
            else:
                # Standardkonfiguration erstellen
                config = self.DEFAULT_CONFIG.copy()
                self.save_config(config)
                self.logger.info("Standardkonfiguration erstellt und gespeichert")
                return config
        except Exception as e:
            self.logger.error(f"Fehler beim Laden der Konfiguration: {str(e)}")
            # Bei Fehlern Standardkonfiguration verwenden
            try:
                self.save_config(self.DEFAULT_CONFIG)
            except Exception as save_error:
                self.logger.error(f"Fehler beim Speichern der Standardkonfiguration: {str(save_error)}")
            return self.DEFAULT_CONFIG.copy()

    def save_config(self, new_config=None):
        """
        Speichert die Konfiguration in der Datei
        
        Args:
            new_config: Neue Konfiguration (optional)
            
        Returns:
            True bei Erfolg, False bei Fehler
        """
        try:
            # Verwende die angegebene Konfiguration oder die aktuelle
            config_to_save = new_config if new_config is not None else self.config

            # Stelle sicher, dass das Verzeichnis existiert
            os.makedirs(os.path.dirname(self.config_path), exist_ok=True)

            # Konfiguration speichern
            with open(self.config_path, 'w') as f:
                json.dump(config_to_save, f, indent=4)

            # Aktuelle Konfiguration aktualisieren
            if new_config is not None:
                self.config = new_config.copy()
                
            self.logger.debug("Konfiguration erfolgreich gespeichert")
            return True
        except Exception as e:
            self.logger.error(f"Fehler beim Speichern der Konfiguration: {str(e)}")
            return False

    def get(self, key, default=None):
        """
        Gibt einen Konfigurationswert zurück
        
        Args:
            key: Durch Punkte getrennter Schlüssel (z.B. 'camera.device_path')
            default: Standardwert, falls Schlüssel nicht gefunden (optional)
            
        Returns:
            Konfigurationswert oder Standardwert
        """
        try:
            # Schlüssel in Teile aufteilen
            parts = key.split('.')

            # Durch verschachteltes Dictionary navigieren
            value = self.config
            for part in parts:
                if part in value:
                    value = value[part]
                else:
                    return default

            return value
        except Exception as e:
            self.logger.error(f"Fehler beim Abrufen des Konfigurationswerts: {str(e)}")
            return default

    def set(self, key, value):
        """
        Setzt einen Konfigurationswert
        
        Args:
            key: Durch Punkte getrennter Schlüssel (z.B. 'camera.device_path')
            value: Zu setzender Wert
            
        Returns:
            True bei Erfolg, False bei Fehler
        """
        try:
            # Schlüssel in Teile aufteilen
            parts = key.split('.')

            # In der aktuellen Konfiguration navigieren
            config = self.config
            for i, part in enumerate(parts[:-1]):
                if part not in config:
                    config[part] = {}
                config = config[part]

            # Wert setzen
            config[parts[-1]] = value

            # Konfiguration speichern
            return self.save_config()
        except Exception as e:
            self.logger.error(f"Fehler beim Setzen des Konfigurationswerts: {str(e)}")
            return False

    def _merge_configs(self, default_config, user_config):
        """
        Führt zwei Konfigurationen zusammen
        
        Args:
            default_config: Standardkonfiguration
            user_config: Benutzerkonfiguration
            
        Returns:
            Zusammengeführte Konfiguration
        """
        result = default_config.copy()
        
        for key, value in user_config.items():
            if key in result and isinstance(result[key], dict) and isinstance(value, dict):
                # Rekursiv zusammenführen, wenn beide Werte Dictionaries sind
                result[key] = self._merge_configs(result[key], value)
            else:
                # Sonst den Wert aus der Benutzerkonfiguration übernehmen
                result[key] = value
                
        return result


# Globale Instanz des ConfigManager erstellen

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
app/utils/secure_path.py
Sicherheitsfunktionen für die Pfadverarbeitung
"""

import os
import string
import logging

logger = logging.getLogger(__name__)

def secure_filename(filename):
    """
    Bereinigt einen Dateinamen und entfernt potenziell gefährliche Zeichen.
    Erlaubt nur alphanumerische Zeichen, Bindestrich, Unterstrich und Punkt.
    
    Args:
        filename: Der zu bereinigende Dateiname
        
    Returns:
        Bereinigter Dateiname ohne gefährliche Zeichen
    """
    if not filename:
        return ""
        
    # Extrahiere nur den Dateinamen ohne Pfad
    filename = os.path.basename(filename)
    
    # Erlaube alphanumerische Zeichen und bestimmte Sonderzeichen
    allowed_chars = set(string.ascii_letters + string.digits + '-_.')
    filename = ''.join(c for c in filename if c in allowed_chars)
    
    # Stellen Sie sicher, dass der Dateiname nicht leer ist
    if not filename:
        logger.warning(f"Dateiname wurde zu einer leeren Zeichenfolge bereinigt")
        return "unnamed_file"
        
    return filename

def secure_join(base_path, *paths):
    """
    Verbindet Pfadsegmente sicher und verhindert Verzeichnistraversierungsangriffe.
    
    Args:
        base_path: Der Basispfad, der nicht verlassen werden soll
        *paths: Zu verbindende Pfadsegmente
        
    Returns:
        Ein sicherer absoluter Pfad, der nicht außerhalb des Basispfads liegt
    """
    # Stelle sicher, dass der Basispfad ein absoluter Pfad ist
    base_path = os.path.abspath(base_path)
    
    # Normalisiere alle Pfadsegmente und entferne potenziell gefährliche Teile
    clean_paths = []
    for i, path in enumerate(paths):
        if i == len(paths) - 1 and path:  # Letztes Element könnte ein Dateiname sein
            clean_paths.append(secure_filename(path))
        else:
            # Normalisiere Pfade und entferne '..' und Anführungszeichen
            if path:
                normalized = os.path.normpath(path).replace('..', '').replace('"', '').replace("'", "")
                clean_paths.append(normalized)
    
    # Verbinde den Basispfad mit den bereinigten Pfadsegmenten
    joined_path = os.path.join(base_path, *clean_paths)
    safe_path = os.path.abspath(joined_path)
    
    # Überprüfe, ob der resultierende Pfad innerhalb des Basispfads liegt
    if not safe_path.startswith(base_path):
        logger.warning(f"Versuchter Pfad-Traversal-Angriff verhindert: {joined_path}")
        return base_path
    
    return safe_path

def is_safe_path(base_path, path):
    """
    Überprüft, ob ein Pfad sicher ist (keine Traversierungsangriffe ermöglicht).
    
    Args:
        base_path: Der Basispfad, der nicht verlassen werden soll
        path: Der zu überprüfende Pfad
        
    Returns:
        True, wenn der Pfad sicher ist, andernfalls False
    """
    # Absolute Pfade erstellen
    base_path = os.path.abspath(base_path)
    test_path = os.path.abspath(os.path.join(base_path, path))
    
    # Prüfen, ob der getestete Pfad innerhalb des Basispfads liegt
    return test_path.startswith(base_path)

def validate_project_id(project_id):
    """
    Validiert eine Projekt-ID gegen Traversierungsangriffe.
    
    Args:
        project_id: Die zu validierende Projekt-ID
        
    Returns:
        True, wenn die Projekt-ID sicher ist, andernfalls False
    """
    if not project_id or not isinstance(project_id, str):
        return False
        
    # Entferne Leerzeichen am Anfang/Ende
    project_id = project_id.strip()
    
    # Überprüfe auf unerlaubte Zeichen oder Sequenzen
    invalid_sequences = ['..', '/', '\\', '~', '$', '&', '*', '(', ')', '`', "'", '"', '|', '<', '>', ';']
    for seq in invalid_sequences:
        if seq in project_id:
            return False
            
    return len(project_id) > 0

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
app/utils/device_detector.py
Modul zur Erkennung von Kameras und Arduino-Geräten
"""

import os
import time
import logging
import subprocess
import glob
import threading
from serial.tools import list_ports

from app.utils.webcam_detection_helper import find_working_webcam

# Logger konfigurieren
logger = logging.getLogger(__name__)

class DeviceDetector:
    """Erkennt und verfolgt verfügbare Kameras und Arduino-Geräte"""

    def __init__(self, scan_interval=10):
        """
        Initialisiert den Gerätedetektor
        
        Args:
            scan_interval: Zeit in Sekunden zwischen Geräte-Scans
        """
        self.scan_interval = scan_interval
        self.devices = {
            'cameras': {
                'webcams': [],
                'gphoto2': []
            },
            'arduinos': []
        }
        self.running = False
        self.lock = threading.Lock()
        self.scan_thread = None
        
        logger.debug(f"DeviceDetector initialisiert mit Scan-Intervall von {scan_interval} Sekunden")
        
        # Sofortiger Scan beim Start
        self.scan_devices()

    def detect_webcams(self):
        """
        Erkennt angeschlossene Webcams
        
        Returns:
            Liste der erkannten Webcams
        """
        webcam_devices = []

        try:
            # Videoeingabegeräte finden
            video_devices = glob.glob('/dev/video*')
            working_webcam = find_working_webcam(video_devices)

            if working_webcam:
                webcam_devices.append(working_webcam)

            for device in video_devices:
                if device != working_webcam and os.path.exists(device):
                    webcam_devices.append(device)
                    
            logger.debug(f"Webcam-Erkennung: {len(webcam_devices)} Geräte gefunden")
        except Exception as e:
            logger.error(f"Fehler bei der Webcam-Erkennung: {e}")

        return webcam_devices

    def detect_gphoto2_cameras(self):
        """
        Erkennt Kameras, die mit gphoto2 kompatibel sind
        
        Returns:
            Liste der erkannten gphoto2-Kameras
        """
        gphoto2_cameras = []

        try:
            # Prüfen, ob gphoto2 installiert ist
            result = subprocess.run(['which', 'gphoto2'],
                                    stdout=subprocess.PIPE,
                                    stderr=subprocess.PIPE)

            if result.returncode == 0:
                # Kameras auflisten
                try:
                    result = subprocess.run(['gphoto2', '--auto-detect'],
                                        stdout=subprocess.PIPE,
                                        stderr=subprocess.PIPE,
                                        text=True,
                                        timeout=5)  # Timeout hinzugefügt

                    if result.returncode == 0:
                        lines = result.stdout.strip().split('\n')
                        for line in lines[2:]:  # Überschriften überspringen
                            if line.strip():
                                gphoto2_cameras.append(line.strip())
                                
                    logger.debug(f"gphoto2-Kamera-Erkennung: {len(gphoto2_cameras)} Geräte gefunden")
                except subprocess.TimeoutExpired:
                    logger.warning("Timeout bei der gphoto2-Kamera-Erkennung")
            else:
                logger.debug("gphoto2 ist nicht installiert")
        except Exception as e:
            logger.error(f"Fehler bei der gphoto2-Kamera-Erkennung: {e}")

        return gphoto2_cameras

    def detect_arduinos(self):
        """
        Erkennt angeschlossene Arduino-Geräte
        
        Returns:
            Liste der erkannten Arduino-Geräte
        """
        arduino_devices = []

        try:
            # Serielle Ports auflisten
            ports = list(list_ports.comports())
            for port in ports:
                if "Arduino" in port.description or "ACM" in port.device:
                    arduino_devices.append({
                        'port': port.device,
                        'description': port.description
                    })
                    
            logger.debug(f"Arduino-Erkennung: {len(arduino_devices)} Geräte gefunden")
        except Exception as e:
            logger.error(f"Fehler bei der Arduino-Erkennung: {e}")

        return arduino_devices

    def scan_devices(self):
        """Scannt nach allen verfügbaren Geräten"""
        with self.lock:
            self.devices['cameras']['webcams'] = self.detect_webcams()
            self.devices['cameras']['gphoto2'] = self.detect_gphoto2_cameras()
            self.devices['arduinos'] = self.detect_arduinos()

        logger.info(f"Geräte erkannt: {len(self.devices['cameras']['webcams'])} Webcams, "
                    f"{len(self.devices['cameras']['gphoto2'])} gphoto2-Kameras, "
                    f"{len(self.devices['arduinos'])} Arduino-Geräte")

    def get_devices(self):
        """
        Gibt die aktuell erkannten Geräte zurück
        
        Returns:
            Dictionary mit erkannten Geräten
        """
        with self.lock:
            devices_copy = {
                'cameras': {
                    'webcams': self.devices['cameras']['webcams'].copy(),
                    'gphoto2': self.devices['cameras']['gphoto2'].copy()
                },
                'arduinos': self.devices['arduinos'].copy()
            }
            return devices_copy

    def _run_detection_loop(self):
        """Ausführungsschleife für den Erkennungsprozess"""
        logger.info("Geräte-Erkennungsschleife gestartet")
        while self.running:
            try:
                self.scan_devices()
                time.sleep(self.scan_interval)
            except Exception as e:
                logger.error(f"Fehler in der Erkennungsschleife: {e}")
                time.sleep(self.scan_interval)  # Trotzdem weiterlaufen

    def start_detection(self):
        """Startet den Erkennungsprozess in einer Schleife"""
        if self.running:
            logger.warning("Erkennungsprozess läuft bereits")
            return
            
        self.running = True
        self.scan_thread = threading.Thread(target=self._run_detection_loop)
        self.scan_thread.daemon = True  # Thread beendet sich, wenn Hauptprogramm beendet
        self.scan_thread.start()
        
        logger.info("Geräte-Erkennung gestartet")

    def stop_detection(self):
        """Stoppt den Erkennungsprozess"""
        self.running = False
        if self.scan_thread and self.scan_thread.is_alive():
            self.scan_thread.join(timeout=2.0)  # Warte maximal 2 Sekunden
        logger.info("Geräte-Erkennung gestoppt")

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
app/utils/viewer_generator.py
Generiert den 360°-Viewer aus einer Serie von Bildern
"""

import os
import json
import shutil
import logging
import time
from datetime import datetime
from PIL import Image
from app.utils.path_manager import PathManager

class ViewerGenerator:
    """Klasse zur Generierung eines 360°-Viewers aus Bildern"""
    
    def __init__(self, photo_dir=None, output_dir=None):
        """
        Initialisiert den Viewer-Generator
        
        Args:
            photo_dir: Verzeichnis mit den Quellfotos (optional)
            output_dir: Ausgabeverzeichnis für generierte Projekte (optional)
        """
        # Logger einrichten
        self.logger = logging.getLogger(__name__)
        
        # Verzeichnisse festlegen
        path_manager = PathManager()
        self.photo_dir = photo_dir or path_manager.photos_dir
        self.output_dir = output_dir or path_manager.projects_dir

        # Sicherstellen, dass das Ausgabeverzeichnis existiert
        os.makedirs(self.output_dir, exist_ok=True)
        self.logger.debug(f"Viewer-Generator initialisiert: {self.photo_dir} -> {self.output_dir}")

    def prepare_images(self, images, project_name):
        """
        Bereitet Bilder für den 360°-Viewer vor (Größenanpassung, Optimierung)
        
        Args:
            images: Liste der Bildpfade
            project_name: Name des Projekts
            
        Returns:
            Pfad zum Projektverzeichnis und Liste der verarbeiteten Bilder
        """
        project_dir = os.path.join(self.output_dir, project_name)
        os.makedirs(project_dir, exist_ok=True)

        processed_images = []

        for i, img_path in enumerate(images):
            try:
                # Bild laden
                full_img_path = os.path.join(self.photo_dir, img_path)
                img = Image.open(full_img_path)

                # Größe anpassen (max. 1200px Breite für optimale Performance)
                max_width = 1200
                if img.width > max_width:
                    ratio = max_width / img.width
                    new_height = int(img.height * ratio)
                    img = img.resize((max_width, new_height), Image.LANCZOS)

                # Optimiertes Bild speichern
                img_filename = f"image_{i:03d}.jpg"
                output_path = os.path.join(project_dir, img_filename)
                img.save(output_path, "JPEG", quality=85, optimize=True)

                processed_images.append(img_filename)
                self.logger.debug(f"Bild {i+1}/{len(images)} verarbeitet: {img_path} -> {img_filename}")
            except Exception as e:
                self.logger.error(f"Fehler bei der Bildverarbeitung für {img_path}: {e}")

        return project_dir, processed_images

    def generate_viewer(self, images, metadata=None):
        """
        Generiert einen 360°-Viewer aus den gegebenen Bildern
        
        Args:
            images: Liste der Bildpfade
            metadata: Zusätzliche Metadaten für das Projekt (optional)
            
        Returns:
            URL zum erstellten Viewer
        """
        if not images:
            self.logger.error("Keine Bilder zum Generieren des Viewers gefunden")
            return None

        # Projektname erstellen (Zeitstempel)
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        project_name = f"project_{timestamp}"

        # Bilder vorbereiten
        project_dir, processed_images = self.prepare_images(images, project_name)

        # Projektmetadaten erstellen
        project_metadata = {
            "name": metadata.get("name", project_name) if metadata else project_name,
            "created": time.time(),
            "image_count": len(processed_images),
            "images": processed_images,
            "user_metadata": metadata or {}
        }

        # Metadaten speichern
        metadata_path = os.path.join(project_dir, "metadata.json")
        with open(metadata_path, "w") as f:
            json.dump(project_metadata, f, indent=4)

        self.logger.info(f"Viewer für Projekt {project_name} mit {len(processed_images)} Bildern erstellt")
        return f"/viewer?project={project_name}"

def find_working_webcam(preferred_devices=None):
    """
    Findet eine funktionierende Webcam
    
    Args:
        preferred_devices: Liste bevorzugter Geräte (optional)
        
    Returns:
        Pfad zum funktionierenden Gerät oder None
    """
    # Bevorzugte Geräte, falls nicht angegeben
    if preferred_devices is None:
        preferred_devices = ['/dev/video0', '/dev/video1']

    # Zuerst fswebcam verwenden, um die Gerätefunktionalität zu testen
    for device in preferred_devices:
        try:
            # fswebcam-Test ausführen
            result = subprocess.run([
                'fswebcam',
                '-d', device,
                '--no-banner',
                '/dev/null'  # Ausgabe verwerfen
            ], capture_output=True, text=True, timeout=3)

            # Wenn fswebcam erfolgreich ist, dieses Gerät zurückgeben
            if result.returncode == 0:
                logger.info(f"Funktionierende Webcam gefunden (fswebcam): {device}")
                return device
        except subprocess.TimeoutExpired:
            logger.debug(f"Zeitüberschreitung bei fswebcam für {device}")
            continue
        except Exception as e:
            logger.debug(f"Fehler bei fswebcam für {device}: {str(e)}")
            continue

    # Fallback auf OpenCV-Erkennung
    for device in preferred_devices:
        try:
            # OpenCV-Capture testen
            cap = cv2.VideoCapture(device)
            ret, frame = cap.read()
            if ret and frame is not None and frame.size > 0:
                cap.release()
                logger.info(f"Funktionierende Webcam gefunden (OpenCV): {device}")
                return device
            cap.release()
        except Exception as e:
            logger.debug(f"Fehler bei OpenCV für {device}: {str(e)}")
            pass

    # Umfassende Suche als letzter Ausweg
    try:
        # v4l2-ctl verwenden, um alle Videogeräte aufzulisten
        result = subprocess.run(['v4l2-ctl', '--list-devices'],
                                capture_output=True,
                                text=True)

        # Alle /dev/video*-Geräte extrahieren
        devices = [
            line.strip()
            for line in result.stdout.split('\n')
            if line.startswith('/dev/video')
        ]

        # Jedes entdeckte Gerät testen
        for device in devices:
            try:
                # Zuerst fswebcam testen
                result = subprocess.run([
                    'fswebcam',
                    '-d', device,
                    '--no-banner',
                    '/dev/null'
                ], capture_output=True, text=True, timeout=3)

                if result.returncode == 0:
                    logger.info(f"Funktionierende Webcam gefunden (umfassende Suche): {device}")
                    return device

                # Alternativ OpenCV testen
                cap = cv2.VideoCapture(device)
                ret, frame = cap.read()
                if ret and frame is not None and frame.size > 0:
                    cap.release()
                    logger.info(f"Funktionierende Webcam gefunden (umfassende Suche): {device}")
                    return device
                cap.release()
            except Exception:
                continue
    except Exception as e:
        logger.error(f"Fehler bei der umfassenden Webcam-Suche: {str(e)}")

    logger.warning("Keine funktionierende Webcam gefunden")
    return None


def get_camera_capabilities(device_path):
    """
    Ruft die Fähigkeiten einer Kamera ab
    
    Args:
        device_path: Pfad zum Kameragerät
        
    Returns:
        Dictionary mit den Fähigkeiten der Kamera
    """
    capabilities = {
        'supported_resolutions': [],
        'max_width': 0,
        'max_height': 0
    }

    try:
        # v4l2-ctl verwenden, um detaillierte Geräteinformationen zu erhalten
        result = subprocess.run([
            'v4l2-ctl',
            '-d', device_path,
            '--list-formats-ext'
        ], capture_output=True, text=True, timeout=3)

        # Ausgabe parsen, um Auflösungen zu extrahieren
        resolutions = []
        for line in result.stdout.split('\n'):
            if 'Size' in line:
                try:
                    # Auflösung wie '640x480' extrahieren
                    resolution = line.split(':')[-1].strip()
                    width, height = map(int, resolution.split('x'))
                    resolutions.append((width, height))

                    # Maximale Auflösung aktualisieren
                    capabilities['max_width'] = max(capabilities['max_width'], width)
                    capabilities['max_height'] = max(capabilities['max_height'], height)
                except Exception as e:
                    logger.debug(f"Fehler beim Parsen der Auflösung: {str(e)}")
                    pass

        capabilities['supported_resolutions'] = sorted(set(resolutions))
    except Exception as e:
        logger.error(f"Fehler beim Abrufen der Kamera-Fähigkeiten: {str(e)}")

    return capabilities


def test_webcam_capture(device_path, width=None, height=None):
    """
    Versucht, ein Bild vom angegebenen Gerät aufzunehmen
    
    Args:
        device_path: Pfad zum Kameragerät
        width: Gewünschte Breite (optional)
        height: Gewünschte Höhe (optional)
        
    Returns:
        Tupel (Erfolg, Pfad zum aufgenommenen Bild)
    """
    try:
        # Ausgabeverzeichnis sicherstellen
        os.makedirs('static/photos', exist_ok=True)

        # Eindeutigen Dateinamen generieren
        filename = f'webcam_test_{int(time.time())}.jpg'
        full_path = os.path.join('static/photos', filename)

        # fswebcam-Befehl vorbereiten
        cmd = ['fswebcam',
               '-d', device_path,
               '--no-banner']

        # Auflösung hinzufügen, falls angegeben
        if width and height:
            cmd.extend(['-r', f'{width}x{height}'])

        cmd.append(full_path)

        # Zuerst fswebcam versuchen
        subprocess.run(cmd, check=True, timeout=5)

        # Prüfen, ob die Datei erstellt wurde
        if os.path.exists(full_path) and os.path.getsize(full_path) > 0:
            logger.info(f"Testbild mit fswebcam aufgenommen: {full_path}")
            return True, full_path

        # Fallback auf OpenCV mit Auflösung
        cap = cv2.VideoCapture(device_path)

        # Auflösung einstellen, falls angegeben
        if width and height:
            cap.set(cv2.CAP_PROP_FRAME_WIDTH, width)
            cap.set(cv2.CAP_PROP_FRAME_HEIGHT, height)

        ret, frame = cap.read()
        cap.release()
        
        if ret:
            cv2.imwrite(full_path, frame)
            logger.info(f"Testbild mit OpenCV aufgenommen: {full_path}")
            return True, full_path

        logger.warning(f"Konnte kein Bild von {device_path} aufnehmen")
        return False, None

    except subprocess.TimeoutExpired:
        logger.error(f"Zeitüberschreitung bei der Aufnahme von {device_path}")
        return False, None
    except Exception as e:
        logger.error(f"Fehler bei der Webcam-Aufnahme: {str(e)}")
        return False, None

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
app/utils/config_manager.py
Verwaltet die Benutzerkonfiguration
"""

import os
import json
import logging

class ConfigManager:
    """Klasse zur Verwaltung der Benutzerkonfiguration"""
    
    DEFAULT_CONFIG = {
        'camera': {
            'device_path': '/dev/video0',
            'type': 'webcam',
            'resolution': {
                'width': 1280,
                'height': 720
            }
        },
        'arduino': {
            'port': '/dev/ttyACM0',
            'baudrate': 9600
        },
        'rotation': {
            'default_degrees': 15,
            'default_interval': 5
        },
        'simulator': {
            'enabled': True
        }
    }

    def __init__(self, config_path=None):
        """
        Initialisiert die Konfigurationsverwaltung
        
        Args:
            config_path: Pfad zur Konfigurationsdatei (optional)
        """
        # Logger einrichten
        self.logger = logging.getLogger(__name__)
        
        # Projektverzeichnis bestimmen
        self.project_dir = os.path.abspath(os.path.dirname(os.path.dirname(os.path.dirname(__file__))))

        # Konfigurationspfad festlegen
        if config_path is None:
            config_path = os.path.join(self.project_dir, 'config.json')
        self.config_path = config_path
        
        # Konfiguration laden
        self.config = self.load_config()
        self.logger.debug(f"ConfigManager initialisiert mit Konfigurationspfad: {self.config_path}")

    def _merge_configs(self, default_config, user_config):
        """
        Führt zwei Konfigurationen zusammen
        
        Args:
            default_config: Standardkonfiguration
            user_config: Benutzerkonfiguration
            
        Returns:
            Zusammengeführte Konfiguration
        """
        result = default_config.copy()
        
        for key, value in user_config.items():
            if key in result and isinstance(result[key], dict) and isinstance(value, dict):
                # Rekursiv zusammenführen, wenn beide Werte Dictionaries sind
                result[key] = self._merge_configs(result[key], value)
            else:
                # Sonst den Wert aus der Benutzerkonfiguration übernehmen
                result[key] = value
                
        return result

    def load_config(self):
        """
        Lädt die Konfiguration aus der Datei oder erstellt die Standardkonfiguration
        
        Returns:
            Konfiguration als Dictionary
        """
        try:
            # Stelle sicher, dass das Verzeichnis existiert
            os.makedirs(os.path.dirname(self.config_path), exist_ok=True)

            # Versuche, bestehende Konfiguration zu laden
            if os.path.exists(self.config_path):
                with open(self.config_path, 'r') as f:
                    config = json.load(f)
                    # Mit Standardkonfiguration zusammenführen, um sicherzustellen,
                    # dass alle Schlüssel existieren
                    merged_config = self._merge_configs(self.DEFAULT_CONFIG, config)
                    self.logger.debug("Konfiguration aus Datei geladen und zusammengeführt")
                    return merged_config
            else:
                # Standardkonfiguration erstellen
                config = self.DEFAULT_CONFIG.copy()
                self.save_config(config)
                self.logger.info("Standardkonfiguration erstellt und gespeichert")
                return config
        except Exception as e:
            self.logger.error(f"Fehler beim Laden der Konfiguration: {str(e)}")
            # Bei Fehlern Standardkonfiguration verwenden
            try:
                self.save_config(self.DEFAULT_CONFIG)
            except Exception as save_error:
                self.logger.error(f"Fehler beim Speichern der Standardkonfiguration: {str(save_error)}")
            return self.DEFAULT_CONFIG.copy()

    def save_config(self, new_config=None):
        """
        Speichert die Konfiguration in der Datei
        
        Args:
            new_config: Neue Konfiguration (optional)
            
        Returns:
            True bei Erfolg, False bei Fehler
        """
        try:
            # Verwende die angegebene Konfiguration oder die aktuelle
            config_to_save = new_config if new_config is not None else self.config

            # Stelle sicher, dass das Verzeichnis existiert
            os.makedirs(os.path.dirname(self.config_path), exist_ok=True)

            # Konfiguration speichern
            with open(self.config_path, 'w') as f:
                json.dump(config_to_save, f, indent=4)

            # Aktuelle Konfiguration aktualisieren
            if new_config is not None:
                self.config = new_config.copy()
                
            self.logger.debug("Konfiguration erfolgreich gespeichert")
            return True
        except Exception as e:
            self.logger.error(f"Fehler beim Speichern der Konfiguration: {str(e)}")
            return False

    def get(self, key, default=None):
        """
        Gibt einen Konfigurationswert zurück
        
        Args:
            key: Durch Punkte getrennter Schlüssel (z.B. 'camera.device_path')
            default: Standardwert, falls Schlüssel nicht gefunden (optional)
            
        Returns:
            Konfigurationswert oder Standardwert
        """
        try:
            # Schlüssel in Teile aufteilen
            parts = key.split('.')

            # Durch verschachteltes Dictionary navigieren
            value = self.config
            for part in parts:
                if part in value:
                    value = value[part]
                else:
                    return default

            return value
        except Exception as e:
            self.logger.error(f"Fehler beim Abrufen des Konfigurationswerts: {str(e)}")
            return default


# Globale Instanz des ConfigManager erstellen

"""
app/utils/jinja_filters.py
Benutzerdefinierte Jinja2-Filter
"""

import datetime

def datetime_filter(value, format='%d.%m.%Y %H:%M'):
    """
    Formatiert einen Timestamp oder datetime-Objekt
    
    Args:
        value: Timestamp oder datetime-Objekt
        format: Ausgabeformat für das Datum
    
    Returns:
        Formatierter Datums-String
    """
    if value is None:
        return ''
    
    # Wenn es sich um einen Timestamp handelt
    if isinstance(value, (int, float)):
        value = datetime.datetime.fromtimestamp(value)
    
    # Wenn es sich um ein datetime-Objekt handelt
    if isinstance(value, datetime.datetime):
        return value.strftime(format)
    
    return str(value)

class CameraController:

    def __init__(self, camera_type='webcam', device='/dev/video0', resolution=None):
        """
        Initialisiert den Kamera-Controller
        
        Args:
            camera_type: Kameratyp ('webcam' oder 'gphoto2') (optional)
            device: Gerätepfad (optional)
            resolution: Auflösung als Tupel (Breite, Höhe) oder String 'BreitexHöhe' (optional)
        """
        self.logger = logging.getLogger(__name__)
        self.camera_type = camera_type
        self.device = device

        # Auflösung als Tupel (Breite, Höhe)
        if isinstance(resolution, str) and 'x' in resolution:
            width, height = resolution.split('x')
            self.resolution = (int(width), int(height))
        elif isinstance(resolution, tuple) and len(resolution) == 2:
            self.resolution = resolution
        elif isinstance(resolution, dict) and 'width' in resolution and 'height' in resolution:
            self.resolution = (resolution['width'], resolution['height'])
        else:
            self.resolution = (1920, 1080)  # Standardauflösung

        self.webcam = None
        self.gphoto2_available = self._check_gphoto2()
        
        self.logger.debug(f"CameraController initialisiert: Typ={self.camera_type}, Gerät={self.device}, "
                        f"Auflösung={self.resolution}")

    def _check_gphoto2(self):
        """
        Prüft, ob gphoto2 installiert ist
        
        Returns:
            True wenn gphoto2 verfügbar ist, sonst False
        """
        try:
            result = subprocess.run(['which', 'gphoto2'],
                                    stdout=subprocess.PIPE,
                                    stderr=subprocess.PIPE)
            return result.returncode == 0
        except Exception as e:
            self.logger.warning(f"Konnte gphoto2 nicht überprüfen: {str(e)}")
            return False

    def _setup_webcam(self):
        """
        Initialisiert die Webcam
        
        Returns:
            True bei Erfolg, False bei Fehler
        """
        if self.webcam is None:
            try:
                # cv2.VideoCapture mit Nummer (0, 1, ...) oder Pfad (/dev/video0, ...)
                if str(self.device).isdigit():
                    device_id = int(self.device)
                else:
                    device_id = self.device

                self.webcam = cv2.VideoCapture(device_id)

                # Auflösung einstellen
                self.webcam.set(cv2.CAP_PROP_FRAME_WIDTH, self.resolution[0])
                self.webcam.set(cv2.CAP_PROP_FRAME_HEIGHT, self.resolution[1])

                if not self.webcam.isOpened():
                    self.logger.error(f"Webcam konnte nicht geöffnet werden: {self.device}")
                    self.webcam = None
                    return False

                self.logger.info(f"Webcam erfolgreich initialisiert: {self.device}")
                return True
            except Exception as e:
                self.logger.error(f"Fehler bei der Webcam-Initialisierung: {str(e)}")
                self.webcam = None
                return False
        return True

    def _close_webcam(self):
        """Schließt die Webcam"""
        if self.webcam is not None:
            self.webcam.release()
            self.webcam = None
            self.logger.debug("Webcam-Verbindung geschlossen")

    def capture_webcam_photo(self, output_path):
        """
        Nimmt ein Foto mit der Webcam auf
        
        Args:
            output_path: Pfad für das aufgenommene Foto
            
        Returns:
            True bei Erfolg, False bei Fehler
        """
        if not self._setup_webcam():
            return False

        try:
            # Stellen Sie sicher, dass das Verzeichnis existiert
            os.makedirs(os.path.dirname(output_path), exist_ok=True)

            # Mehrere Frames lesen, um sicherzustellen, dass die Kamera sich angepasst hat
            for _ in range(5):
                ret, frame = self.webcam.read()
                if not ret:
                    self.logger.error("Fehler beim Lesen des Webcam-Frames")
                    return False
                time.sleep(0.1)

            # Letzten Frame speichern
            success = cv2.imwrite(output_path, frame)

            if success:
                self.logger.info(f"Webcam-Foto gespeichert: {output_path}")
                return True
            else:
                self.logger.error(f"Fehler beim Speichern des Webcam-Fotos: {output_path}")
                return False
        except Exception as e:
            self.logger.error(f"Fehler bei der Webcam-Fotoaufnahme: {str(e)}")
            return False
        finally:
            # Webcam wieder freigeben
            self._close_webcam()

    def capture_gphoto2_photo(self, output_path):
        """
        Nimmt ein Foto mit einer gphoto2-kompatiblen Kamera auf
        
        Args:
            output_path: Pfad für das aufgenommene Foto
            
        Returns:
            True bei Erfolg, False bei Fehler
        """
        if not self.gphoto2_available:
            self.logger.error("gphoto2 ist nicht installiert oder nicht verfügbar")
            return False

        try:
            # Stellen Sie sicher, dass das Verzeichnis existiert
            os.makedirs(os.path.dirname(output_path), exist_ok=True)

            # Zusammengesetzter gphoto2-Befehl
            cmd = [
                'gphoto2',
                '--capture-image-and-download',
                '--filename', output_path
            ]

            # Wenn eine bestimmte Kamera ausgewählt wurde (Port)
            if self.device and self.device != 'auto':
                cmd.extend(['--port', self.device])

            # gphoto2-Befehl ausführen
            self.logger.debug(f"gphoto2-Befehl: {' '.join(cmd)}")
            result = subprocess.run(cmd, capture_output=True, text=True)

            if result.returncode == 0:
                self.logger.info(f"gphoto2-Foto gespeichert: {output_path}")
                return True
            else:
                self.logger.error(f"gphoto2-Fehler: {result.stderr}")
                return False
        except Exception as e:
            self.logger.error(f"Fehler bei der gphoto2-Fotoaufnahme: {str(e)}")
            return False

    def capture_photo(self, output_path):

        Nimmt ein Foto auf (je nach Kameratyp)

            return False

        self._close_webcam()

import time

# Logger einrichten
logger = logging.getLogger(__name__)

class ArduinoController:
    """Controller für die Kommunikation mit dem Arduino, der den Drehteller steuert"""

    def __init__(self, port=None, baudrate=9600):
        """
        Initialisiert die Arduino-Verbindung
        
        Args:
            port: Serieller Port (optional)
            baudrate: Baudrate (optional)
        """
        self.logger = logging.getLogger(__name__)
        self.port = port
        self.baudrate = baudrate
        self.serial = None
        self.connected = False

        # Verbindung herstellen, wenn ein Port angegeben wurde
        if port:
            self.connect()
            
        self.logger.debug(f"ArduinoController initialisiert: Port={self.port}, Baudrate={self.baudrate}")

    def connect(self):
        """
        Stellt eine Verbindung zum Arduino her
        
        Returns:
            True bei Erfolg, False bei Fehler
        """
        try:
            self.serial = serial.Serial(self.port, self.baudrate, timeout=2)
            time.sleep(2)  # Warten auf Arduino-Reset nach Verbindungsaufbau
            self.connected = True
            self.logger.info(f"Verbindung zum Arduino hergestellt: {self.port} @ {self.baudrate} Baud")
            return True
        except Exception as e:
            self.logger.error(f"Fehler beim Verbinden mit Arduino: {str(e)}")
            self.connected = False
            return False

    def disconnect(self):
        """
        Trennt die Verbindung zum Arduino
        
        Returns:
            True bei Erfolg, False bei Fehler
        """
        if self.serial and self.connected:
            try:
                self.serial.close()
                self.connected = False
                self.logger.info("Verbindung zum Arduino getrennt")
                return True
            except Exception as e:
                self.logger.error(f"Fehler beim Trennen der Arduino-Verbindung: {str(e)}")
                return False
        return True  # War nicht verbunden

    def is_connected(self):
        """
        Prüft, ob die Verbindung zum Arduino hergestellt ist
        
        Returns:
            True wenn verbunden, sonst False
        """
        return self.connected

    def send_command(self, command):
        """
        Sendet einen Befehl an den Arduino
        
        Args:
            command: Zu sendender Befehl
            
        Returns:
            True bei Erfolg, False bei Fehler
        """
        if not self.connected:
            if not self.connect():
                self.logger.error("Kann Befehl nicht senden: Keine Verbindung zum Arduino")
                return False

        try:
            # Befehl senden
            self.serial.write(f"{command}\n".encode())

            # Auf Antwort warten
            response = self.serial.readline().decode().strip()
            self.logger.debug(f"Arduino-Antwort: {response}")

            return response == "OK"
        except Exception as e:
            self.logger.error(f"Fehler beim Senden des Befehls an Arduino: {str(e)}")
            # Bei Fehler Verbindung trennen und neu verbinden
            self.disconnect()
            return False

    def turn_motor_on(self):
        """
        Schaltet den Motor ein (Relais schließen)
        
        Returns:
            True bei Erfolg, False bei Fehler
        """
        success = self.send_command("1")
        if success:
            self.logger.debug("Motor eingeschaltet")
        return success

    def turn_motor_off(self):
        """
        Schaltet den Motor aus (Relais öffnen)
        
        Returns:
            True bei Erfolg, False bei Fehler
        """
        success = self.send_command("0")
        if success:
            self.logger.debug("Motor ausgeschaltet")
        return success


def rotate_for_duration(self, duration_ms):
    """
    Dreht den Motor für eine bestimmte Zeit
    
    Args:
        duration_ms: Dauer in Millisekunden
        
    Returns:
        True bei Erfolg, False bei Fehler
    """
    # Validierung der Eingabe
    try:
        duration_ms = float(duration_ms)
        if duration_ms <= 0 or duration_ms > 60000:  # Maximal 1 Minute
            self.logger.error(f"Ungültige Drehzeit: {duration_ms} ms")
            return False
    except (TypeError, ValueError):
        self.logger.error(f"Ungültiger Wert für Drehzeit: {duration_ms}")
        return False
    
    if not self.turn_motor_on():
        return False

    # Warten für die angegebene Dauer
    time.sleep(duration_ms / 1000.0)

    # Motor ausschalten
    success = self.turn_motor_off()
    if success:
        self.logger.debug(f"Motor für {duration_ms} ms gedreht")
    return success

class TurntableController:
    """Controller für die Drehteller-Steuerung mit dem Arduino"""

    # Konstanten für den Drehteller (0,8° pro Sekunde bei diesem Motor)
    MOTOR_DEGREE_PER_SECOND = 0.8

    def __init__(self, arduino_controller, default_angle_step=5):
        """
        Initialisiert den Drehteller-Controller
        
        Args:
            arduino_controller: ArduinoController-Instanz
            default_angle_step: Standard-Winkelschritt in Grad (optional)
        """
        self.logger = logging.getLogger(__name__)
        self.arduino = arduino_controller
        self.default_angle_step = default_angle_step
        self.current_position = 0  # Aktuelle Position in Grad (0-360)
        
        self.logger.debug(f"TurntableController initialisiert: Winkelschritt={self.default_angle_step}°")

    def calculate_rotation_time(self, degrees):
        """
        Berechnet die Zeit, die für eine Rotation um einen bestimmten Winkel benötigt wird
        
        Args:
            degrees: Winkel in Grad
            
        Returns:
            Zeit in Millisekunden
        """
        # Die Zeit wird in Millisekunden zurückgegeben
        return int((degrees / self.MOTOR_DEGREE_PER_SECOND) * 1000)

    def move_degrees(self, degrees):
        """
        Bewegt den Drehteller um einen bestimmten Winkel
        
        Args:
            degrees: Winkel in Grad
            
        Returns:
            True bei Erfolg, False bei Fehler
        """
        if not self.arduino or not self.arduino.is_connected():
            self.logger.error("Arduino ist nicht verbunden")
            return False

        # Rotation Zeit berechnen
        rotation_time_ms = self.calculate_rotation_time(degrees)

        # Motor für die berechnete Zeit einschalten
        success = self.arduino.rotate_for_duration(rotation_time_ms)

        if success:
            # Position aktualisieren
            self.current_position = (self.current_position + degrees) % 360
            self.logger.info(f"Drehteller um {degrees}° gedreht, neue Position: {self.current_position}°")
        else:
            self.logger.error(f"Fehler beim Drehen des Tellers um {degrees}°")

        return success

    def reset_position(self):
        """
        Setzt die aktuelle Position auf 0 Grad zurück (ohne Bewegung)
        
        Returns:
            True
        """
        self.current_position = 0
        self.logger.info("Drehteller-Position zurückgesetzt")
        return True

    def start_session(self, project, camera_controller):
        """
        Startet eine Fotosession für ein Projekt
        
        Args:
            project: Project-Instanz
            camera_controller: CameraController-Instanz
            
        Returns:
            True bei Erfolg, False bei Fehler
        """
        if not self.arduino or not self.arduino.is_connected():

            return False

        if not camera_controller:
            self.logger.error("Kamera-Controller ist nicht initialisiert")
            return False

            # Neue Session erstellen
            session_id = str(uuid.uuid4())
            session = PhotoSession(
                id=session_id,
                name=f"Session {time.strftime('%Y-%m-%d %H:%M')}",
                timestamp=time.time(),
                angle_step=project.angle_step
            )

            # Speicherpfad für die Fotos
            base_path = os.path.join(project.path, "sessions", session_id)
            os.makedirs(base_path, exist_ok=True)

            # Drehteller auf Position 0 zurücksetzen (ohne Bewegung)
            self.reset_position()

            # Anzahl der benötigten Schritte berechnen
            total_steps = math.floor(360 / project.angle_step)

            self.logger.info(f"Starte Fotosession mit {total_steps} Schritten alle {project.angle_step}°")

            for step in range(total_steps):
                # Aktuelle Winkelposition
                angle = step * project.angle_step

                # Dateiname für das Foto
                photo_filename = os.path.join(base_path, f"angle_{angle:03d}.jpg")

                # Foto aufnehmen
                self.logger.info(f"Nehme Foto bei {angle}° auf")
                if not camera_controller.capture_photo(photo_filename):
                    self.logger.error(f"Fehler beim Aufnehmen des Fotos bei {angle}°")
                    return False

                # Session-Informationen aktualisieren
                session.add_photo(angle, photo_filename)

                # Wenn wir nicht beim letzten Schritt sind, drehen wir weiter
                if step < total_steps - 1:
                    self.move_degrees(project.angle_step)
                    # Kurze Pause für Stabilisierung
                    time.sleep(1)

            # Session als abgeschlossen markieren
            session.completed = True
            
            # Session zum Projekt hinzufügen
            project.add_session(session)

            self.logger.info(f"Fotosession erfolgreich abgeschlossen: {session_id}")
            return True

        except Exception as e:
            self.logger.error(f"Fehler während der Fotosession: {str(e)}")
            return False

        self.camera_type = camera_type
        self.device = device

        # Auflösung als Tupel (Breite, Höhe)
        if isinstance(resolution, str) and 'x' in resolution:
            width, height = resolution.split('x')
            self.resolution = (int(width), int(height))
        elif isinstance(resolution, tuple) and len(resolution) == 2:
            self.resolution = resolution
        elif isinstance(resolution, dict) and 'width' in resolution and 'height' in resolution:
            self.resolution = (resolution['width'], resolution['height'])
        else:
            self.resolution = (1920, 1080)  # Standardauflösung

        self.webcam = None
        self.gphoto2_available = self._check_gphoto2()
        
        self.logger.debug(f"CameraController initialisiert: Typ={self.camera_type}, Gerät={self.device}, "
                        f"Auflösung={self.resolution}")

    def _check_gphoto2(self):
        """
        Prüft, ob gphoto2 installiert ist
        
        Returns:
            True wenn gphoto2 verfügbar ist, sonst False
        """
        try:
            result = subprocess.run(['which', 'gphoto2'],
                                    stdout=subprocess.PIPE,
                                    stderr=subprocess.PIPE)
            return result.returncode == 0
        except Exception as e:
            self.logger.warning(f"Konnte gphoto2 nicht überprüfen: {str(e)}")
            return False

    def _setup_webcam(self):
        """
        Initialisiert die Webcam
        
        Returns:
            True bei Erfolg, False bei Fehler
        """
        if self.webcam is None:
            try:
                # cv2.VideoCapture mit Nummer (0, 1, ...) oder Pfad (/dev/video0, ...)
                if str(self.device).isdigit():
                    device_id = int(self.device)
                else:
                    device_id = self.device

                self.webcam = cv2.VideoCapture(device_id)

                # Auflösung einstellen
                self.webcam.set(cv2.CAP_PROP_FRAME_WIDTH, self.resolution[0])
                self.webcam.set(cv2.CAP_PROP_FRAME_HEIGHT, self.resolution[1])

                if not self.webcam.isOpened():
                    self.logger.error(f"Webcam konnte nicht geöffnet werden: {self.device}")
                    self.webcam = None
                    return False

                self.logger.info(f"Webcam erfolgreich initialisiert: {self.device}")
                return True
            except Exception as e:
                self.logger.error(f"Fehler bei der Webcam-Initialisierung: {str(e)}")
                self.webcam = None
                return False
        return True

    def _close_webcam(self):
        """Schließt die Webcam"""
        if self.webcam is not None:
            self.webcam.release()
            self.webcam = None
            self.logger.debug("Webcam-Verbindung geschlossen")

    def capture_webcam_photo(self, output_path):
        """
        Nimmt ein Foto mit der Webcam auf
        
        Args:
            output_path: Pfad für das aufgenommene Foto
            
        Returns:
            True bei Erfolg, False bei Fehler
        """
        if not self._setup_webcam():
            return False

        try:
            # Stellen Sie sicher, dass das Verzeichnis existiert
            os.makedirs(os.path.dirname(output_path), exist_ok=True)

            # Mehrere Frames lesen, um sicherzustellen, dass die Kamera sich angepasst hat
            for _ in range(5):
                ret, frame = self.webcam.read()
                if not ret:
                    self.logger.error("Fehler beim Lesen des Webcam-Frames")
                    return False
                time.sleep(0.1)

            # Letzten Frame speichern
            success = cv2.imwrite(output_path, frame)

            if success:
                self.logger.info(f"Webcam-Foto gespeichert: {output_path}")
                return True
            else:
                self.logger.error(f"Fehler beim Speichern des Webcam-Fotos: {output_path}")
                return False
        except Exception as e:
            self.logger.error(f"Fehler bei der Webcam-Fotoaufnahme: {str(e)}")
            return False
        finally:
            # Webcam wieder freigeben
            self._close_webcam()

    def capture_gphoto2_photo(self, output_path):
        """
        Nimmt ein Foto mit einer gphoto2-kompatiblen Kamera auf
        
        Args:
            output_path: Pfad für das aufgenommene Foto
            
        Returns:
            True bei Erfolg, False bei Fehler
        """
        if not self.gphoto2_available:
            self.logger.error("gphoto2 ist nicht installiert oder nicht verfügbar")
            return False

        try:
            # Stellen Sie sicher, dass das Verzeichnis existiert
            os.makedirs(os.path.dirname(output_path), exist_ok=True)

            # Zusammengesetzter gphoto2-Befehl
            cmd = [
                'gphoto2',
                '--capture-image-and-download',
                '--filename', output_path
            ]

            # Wenn eine bestimmte Kamera ausgewählt wurde (Port)
            if self.device and self.device != 'auto':
                cmd.extend(['--port', self.device])

            # gphoto2-Befehl ausführen
            self.logger.debug(f"gphoto2-Befehl: {' '.join(cmd)}")
            result = subprocess.run(cmd, capture_output=True, text=True)

            if result.returncode == 0:
                self.logger.info(f"gphoto2-Foto gespeichert: {output_path}")
                return True
            else:
                self.logger.error(f"gphoto2-Fehler: {result.stderr}")
                return False
        except Exception as e:
            self.logger.error(f"Fehler bei der gphoto2-Fotoaufnahme: {str(e)}")
            return False

    # Diese Methode fehlte - sie ist die öffentliche Methode, die von außen aufgerufen wird

                return False
        except Exception as e:
            self.logger.error(f"Allgemeiner Fehler bei der Fotoaufnahme: {str(e)}")

def secure_filename(filename):
    """Bereinigt einen Dateinamen und entfernt potenziell gefährliche Zeichen."""
    if filename:
        filename = os.path.basename(filename)
        allowed_chars = set(string.ascii_letters + string.digits + '-_.')
        filename = ''.join(c for c in filename if c in allowed_chars)
    return filename

def secure_join(base_path, *paths):
    """Verbindet Pfadsegmente sicher und verhindert Verzeichnistraversierungsangriffe."""
    base_path = os.path.abspath(base_path)
    safe_path = os.path.abspath(os.path.join(base_path, *[secure_filename(p) if i == len(paths)-1 else p for i, p in enumerate(paths)]))
    
    if not safe_path.startswith(base_path):
        return base_path
        
    return safe_path

    def __init__(self, id=None, name="Neue Session", timestamp=None, angle_step=5):
        """
        Initialisiert eine Fotosession
        
        Args:
            id: Session-ID (optional)
            name: Sessionname (optional)
            timestamp: Zeitstempel (optional)
            angle_step: Winkelschritt in Grad (optional)
        """
        self.id = id or str(uuid.uuid4())
        self.name = name
        self.timestamp = timestamp or time.time()
        self.angle_step = angle_step
        self.photos = {}  # Dictionary mit Winkel als Schlüssel und Dateipfad als Wert
        self.completed = False
        
        logger.debug(f"PhotoSession erstellt: {self.name} (ID: {self.id})")

    def add_photo(self, angle, photo_path):
        """
        Fügt ein Foto zur Session hinzu
        
        Args:
            angle: Winkel in Grad
            photo_path: Pfad zum Foto
        """
        self.photos[angle] = photo_path
        logger.debug(f"Foto bei {angle}° zur Session {self.id} hinzugefügt: {photo_path}")

    def get_photo(self, angle):
        """
        Gibt den Pfad eines Fotos für einen bestimmten Winkel zurück
        
        Args:
            angle: Winkel in Grad
            
        Returns:
            Pfad zum Foto oder None
        """
        return self.photos.get(angle)

    def get_all_photos(self):
        """
        Gibt alle Fotos der Session zurück
        
        Returns:
            Liste von Fotopfaden, sortiert nach Winkel
        """
        # Sortieren nach Winkel
        return [self.photos[angle] for angle in sorted(self.photos.keys())]

    def to_dict(self):
        """
        Konvertiert die Session in ein Dictionary
        
        Returns:
            Dictionary-Repräsentation der Session
        """
        return {
            'id': self.id,
            'name': self.name,
            'timestamp': self.timestamp,
            'angle_step': self.angle_step,
            'photos': self.photos,
            'completed': self.completed
        }

    @classmethod
    def from_dict(cls, data):
        """
        Erstellt eine Session aus einem Dictionary
        
        Args:
            data: Dictionary mit Sessiondaten
            
        Returns:
            PhotoSession-Instanz
        """
        session = cls(
            id=data.get('id'),
            name=data.get('name', "Unbenannte Session"),
            timestamp=data.get('timestamp', time.time()),
            angle_step=data.get('angle_step', 5)
        )

        # String-Schlüssel in Integer umwandeln
        photos = data.get('photos', {})
        if isinstance(photos, dict):
            for angle, path in photos.items():
                try:
                    session.photos[int(angle)] = path
                except ValueError:
                    # Falls der Winkel kein gültiger Integer ist
                    session.photos[angle] = path
        
        session.completed = data.get('completed', False)

        return session

def rotate_teller(degrees):
    """
    Dreht den Teller um den angegebenen Winkel
    
    Args:
        degrees: Drehwinkel in Grad
    """
    # Prüfen, ob Simulator aktiviert ist
    USE_SIMULATOR = config_manager.get('simulator.enabled', True)
    
    if USE_SIMULATOR:
        logger.info(f"Simulierte Rotation: {degrees} Grad")
        return
    
    if arduino_controller is None or not arduino_controller.is_connected():
        logger.error("Arduino nicht verbunden!")
        return
    
    # Berechnung der Drehzeit basierend auf der Gradzahl (0,8 Grad pro Sekunde)
    rotation_time = degrees / 0.8
    
    # Relais einschalten (Drehteller starten)
    arduino_controller.turn_motor_on()
    time.sleep(rotation_time)  # Warte für die berechnete Zeit
    arduino_controller.turn_motor_off()
    
    logger.info(f"Drehteller um {degrees} Grad gedreht.")

def take_photo(project_id, filename=None):

        project_id: ID des Projekts, in dem das Foto gespeichert werden soll
        filename: Optionaler Dateiname

        Pfad zum gespeicherten Foto
    """
    # Prüfen, ob Simulator aktiviert ist
    USE_SIMULATOR = config_manager.get('simulator.enabled', True)
    
    try:
        # Stelle sicher, dass nur der Dateiname verwendet wird
        if filename:
            filename = os.path.basename(filename)
        else:
            # Generiere einen Dateinamen, wenn keiner angegeben wurde
            timestamp = int(time.time())
            filename = f'photo_{timestamp}.jpg'
            
        # Projektverzeichnis ermitteln
        from app.utils.path_manager import PathManager
        path_manager = PathManager()
        
        # Pfad zum Projektverzeichnis erstellen
        project_path = path_manager.get_project_path(project_id)
        
        # Stelle sicher, dass das Bilder-Verzeichnis im Projekt existiert
        images_dir = os.path.join(project_path, 'images')
        os.makedirs(images_dir, exist_ok=True)
        
        # Vollständiger Pfad zum Foto
        full_path = os.path.join(images_dir, filename)
        
        if USE_SIMULATOR:
            # Webcam-Simulator verwenden
            from app.utils.webcam_simulator import webcam_simulator
            photo_path = webcam_simulator.capture_photo(full_path)
            logger.info(f"Simuliertes Foto aufgenommen und als {photo_path} gespeichert.")

        # Reale Kamera verwenden
        if camera_controller:
            success = camera_controller.capture_photo(full_path)
            if success:
                logger.info(f"Foto aufgenommen und als {filename} gespeichert.")
                return full_path
            else:
                logger.error(f"Fehler beim Aufnehmen des Fotos mit dem Kamera-Controller.")
                return None
        else:
            logger.error("Kein Kamera-Controller verfügbar.")
            return None
    except Exception as e:
        logger.error(f"Fehler beim Aufnehmen des Fotos: {e}")
        return None

def secure_filename(filename):
    """Bereinigt einen Dateinamen und entfernt potenziell gefährliche Zeichen."""
    if filename:
        filename = os.path.basename(filename)
        allowed_chars = set(string.ascii_letters + string.digits + '-_.')
        filename = ''.join(c for c in filename if c in allowed_chars)
    return filename

def secure_join(base_path, *paths):
    """Verbindet Pfadsegmente sicher und verhindert Verzeichnistraversierungsangriffe."""
    base_path = os.path.abspath(base_path)
    safe_path = os.path.abspath(os.path.join(base_path, *[secure_filename(p) if i == len(paths)-1 else p for i, p in enumerate(paths)]))
    
    if not safe_path.startswith(base_path):
        return base_path
        
    return safe_path

@projects_bp.route('/<project_id>')
def view_project(project_id):
    """
    Zeigt ein Projekt an
    
    Args:
        project_id: ID des Projekts
    """
    try:
        # Projekt laden
        project = project_manager.get_project(project_id)
        
        if not project:
            # Wenn das Projekt nicht gefunden wurde, zur Projektliste weiterleiten
            logger.warning(f"Projekt nicht gefunden: {project_id}")
            return redirect(url_for('projects.projects_page'))
        
        # Alle Bilder für das Projekt sammeln
        project_images = []
        for session in project.sessions:
            for angle, photo_path in session.photos.items():
                # Stelle sicher, dass der Pfad relativ ist
                try:
                    # Versuche, einen relativen Pfad zu erstellen
                    relative_path = os.path.relpath(
                        photo_path, 
                        start=os.path.join(path_manager.base_dir, 'static')
                    )
                    
                    # Korrigiere Pfad-Separator
                    relative_path = relative_path.replace(os.sep, '/')
                    
                    project_images.append({
                        'angle': angle,
                        'path': relative_path
                    })
                except Exception as path_error:
                    logger.warning(f"Fehler bei Bildpfadverarbeitung: {path_error}")
        
        # Projekt anzeigen
        return render_template('projects/view.html', 
                               project=project, 
                               images=project_images)
    except Exception as e:
        logger.error(f"Fehler beim Anzeigen des Projekts {project_id}: {e}")
        return render_template('error.html', 
                               message=f"Fehler beim Anzeigen des Projekts: {str(e)}"), 500

@projects_bp.route('/image/<path:image_path>')
def serve_project_image(image_path):
    """
    Liefert ein Projektbild aus
    
    Args:
        image_path: Relativer Pfad zum Bild
    """
    try:
        # Bereinige den Bildpfad
        clean_image_path = sanitize_project_image_path(image_path)
        
        # Basis-Verzeichnis für statische Dateien
        static_dir = os.path.join(path_manager.base_dir, 'static')
        
        # Vollständiger Pfad zur Bilddatei
        full_path = os.path.normpath(os.path.join(static_dir, clean_image_path))
        
        # Sicherheitsüberprüfung: Stelle sicher, dass der Pfad innerhalb des statischen Verzeichnisses liegt
        if not full_path.startswith(static_dir):
            logger.warning(f"Ungültiger Bildpfad angefordert: {full_path}")
            abort(403)  # Verboten
        
        # Sicherstellen, dass die Datei existiert
        if not os.path.exists(full_path):
            logger.warning(f"Bild nicht gefunden: {full_path}")
            
            # Versuche alternative Pfade
            alternative_paths = [
                os.path.join(path_manager.base_dir, 'static', 'projects', clean_image_path),
                os.path.join(path_manager.base_dir, 'static', clean_image_path)
            ]
            
            for alt_path in alternative_paths:
                if os.path.exists(alt_path):
                    full_path = alt_path
                    break
            else:
                logger.error(f"Kein alternatives Bild gefunden für: {clean_image_path}")
                abort(404)
        
        # Verzeichnis und Dateiname extrahieren
        directory = os.path.dirname(full_path)
        filename = os.path.basename(full_path)
        
        # Bild senden
        return send_from_directory(directory, filename)
    except Exception as e:
        logger.error(f"Fehler beim Bereitstellen des Bildes: {e}")
        return render_template('error.html', 
                               message=f"Fehler beim Laden des Bildes: {str(e)}"), 500

                const formData = new FormData(settingsForm);

                // FormData in verschachtelte Objekte umwandeln
                formData.forEach((value, key) => {
                    // Checkbox-Werte in boolesche Werte umwandeln
                    if (key === 'simulator.enabled') {
                        value = document.getElementById('simulatorEnabled').checked;
                    }
                    
                    // Zahlen in numerische Werte umwandeln

                    }
                    
                    // Punkt-Notation in verschachtelte Objekte umwandeln
                    const keys = key.split('.');
                    let current = configData;
                    
                    for (let i = 0; i < keys.length - 1; i++) {
                        if (!current[keys[i]]) {
                            current[keys[i]] = {};
                        }
                        current = current[keys[i]];

from app.blueprints.settings import settings_bp
from app.utils.config_manager import config_manager

logger = logging.getLogger(__name__)

def settings():
    """Zeigt die Einstellungsseite an"""
    return render_template('settings.html')

@settings_bp.route('/get_config')
def get_config():
    """Gibt die aktuelle Konfiguration zurück"""
    return jsonify(config_manager.config)

@settings_bp.route('/save_config', methods=['POST'])
def save_config():
    """Speichert die Konfiguration"""
    try:
        # Prüfen, ob JSON-Daten empfangen wurden
        if not request.is_json:
            return jsonify({
                "status": "error", 
                "message": "Keine JSON-Daten empfangen. Content-Type muss application/json sein."
            }), 400
        
        # JSON-Daten abrufen
        new_config = request.json
        
        if not new_config:
            return jsonify({
                "status": "error", 
                "message": "Leere Konfigurationsdaten empfangen"
            }), 400
        
        # Konfigurationsdaten ausgeben für Debugging
        logger.debug(f"Empfangene Konfiguration: {json.dumps(new_config, indent=2)}")
        
        # Rekursive Aktualisierung der Konfiguration
        def recursive_update(target, source):
            for key, value in source.items():
                if isinstance(value, dict) and key in target and isinstance(target[key], dict):
                    recursive_update(target[key], value)
                else:
                    target[key] = value
        
        # Kopie der aktuellen Konfiguration erstellen
        merged_config = config_manager.config.copy()
        
        # Neue Konfiguration rekursiv zusammenführen
        recursive_update(merged_config, new_config)
        
        # Versuchen, die Konfiguration zu speichern
        config_save_success = config_manager.save_config(merged_config)
        
        if config_save_success:
            logger.info(f"Konfiguration erfolgreich gespeichert: {config_manager.config_path}")
            return jsonify({
                "status": "success", 
                "message": "Konfiguration erfolgreich gespeichert",
                "config": merged_config
            })
        else:
            return jsonify({
                "status": "error", 
                "message": "Fehler beim Speichern der Konfiguration"
            }), 500
            
    except Exception as e:
        logger.error(f"Allgemeiner Fehler beim Speichern der Konfiguration: {e}")
        return jsonify({
            "status": "error", 
            "message": f"Allgemeiner Fehler: {str(e)}"
        }), 500

                if isinstance(value, dict) and key in target and isinstance(target[key], dict):
                    recursive_update(target[key], value)
                else:
                    target[key] = value
        
        # Kopie der aktuellen Konfiguration erstellen
        merged_config = config_manager.config.copy()
        
        # Neue Konfiguration rekursiv zusammenführen
        recursive_update(merged_config, new_config)
        
        # Versuchen, die Konfiguration zu speichern
        config_save_success = config_manager.save_config(merged_config)
        
        if config_save_success:
            logger.info(f"Konfiguration erfolgreich gespeichert: {config_manager.config_path}")
            return jsonify({
                "status": "success", 
                "message": "Konfiguration erfolgreich gespeichert",
                "config": merged_config
            })
        else:
            return jsonify({
                "status": "error", 
                "message": "Fehler beim Speichern der Konfiguration"
            }), 500
            
    except Exception as e:
        logger.error(f"Allgemeiner Fehler beim Speichern der Konfiguration: {e}")
        return jsonify({
            "status": "error", 
            "message": f"Allgemeiner Fehler: {str(e)}"
        }), 500
